add_library(mininids_platform desktop/CommandLine.cpp)
target_sources(mininids_platform PRIVATE desktop/LiveSource.cpp)
target_include_directories(mininids_platform PUBLIC include)
target_link_libraries(mininids_platform PUBLIC mininids_options mininids_core ${CMAKE_DL_LIBS})
if(WIN32)
  target_link_libraries(mininids_platform PRIVATE shell32)
endif()
if(UNIX AND NOT APPLE)
  find_library(MININIDS_CAPTURE_RUNTIME NAMES pcap)
  target_compile_definitions(mininids_platform PRIVATE MININIDS_CAPTURE_LIBDIR="${CMAKE_INSTALL_LIBDIR}")
  if(MININIDS_CAPTURE_RUNTIME)
    get_filename_component(capture_runtime "${MININIDS_CAPTURE_RUNTIME}" REALPATH)
    install(FILES "${capture_runtime}" DESTINATION ${CMAKE_INSTALL_LIBDIR} RENAME libpcap.so.1)
  endif()
endif()
