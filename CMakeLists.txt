cmake_minimum_required(VERSION 3.25)
project(MiniNIDS VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(MININIDS_BUILD_GUI "Build the Qt Quick application" OFF)
option(MININIDS_BUILD_TESTS "Build GoogleTest tests" ON)
option(MININIDS_BUILD_BENCHMARKS "Build Google Benchmark suite" OFF)
option(MININIDS_BUILD_FUZZERS "Build Clang libFuzzer targets" OFF)
option(MININIDS_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(MININIDS_FETCH_DEPS "Fetch pinned sources if packages are missing" ON)
option(MININIDS_DEPLOY_RUNTIME "Bundle GUI runtime dependencies during installation" ON)
set(MININIDS_PACKAGE_LABEL "${PROJECT_VERSION}" CACHE STRING "Release asset version label")
set(MININIDS_PACKAGE_REVISION "0" CACHE STRING "Monotonic development package revision")
if(NOT MININIDS_PACKAGE_REVISION MATCHES "^[0-9]+$")
  message(FATAL_ERROR "Invalid MININIDS_PACKAGE_REVISION")
endif()
if(NOT MININIDS_PACKAGE_LABEL MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+([-+][A-Za-z0-9.-]+)?$")
  message(FATAL_ERROR "Invalid MININIDS_PACKAGE_LABEL")
endif()
if(WIN32)
  set(CMAKE_INSTALL_BINDIR "." CACHE PATH "Application executables" FORCE)
endif()
include(GNUInstallDirs)
if(APPLE AND MININIDS_BUILD_GUI)
  set(MININIDS_DATA_DIR "Mini-NIDS.app/Contents/Resources/share/mininids")
else()
  set(MININIDS_DATA_DIR "${CMAKE_INSTALL_DATADIR}/mininids")
endif()
include(cmake/CompilerOptions.cmake)
include(cmake/Sanitizers.cmake)
include(cmake/Dependencies.cmake)
if(MININIDS_BUILD_TESTS)
  enable_testing()
endif()
add_subdirectory(core)
add_subdirectory(platform)
add_subdirectory(apps/cli)
if(MININIDS_BUILD_GUI)
  add_subdirectory(apps/gui)
endif()
if(MININIDS_BUILD_TESTS)
  add_subdirectory(tests)
endif()
if(MININIDS_BUILD_BENCHMARKS)
  add_subdirectory(benchmarks)
endif()
if(MININIDS_BUILD_FUZZERS)
  add_subdirectory(fuzz)
endif()
include(cmake/Packaging.cmake)
