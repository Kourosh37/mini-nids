add_executable(mininids main.cpp)
target_compile_definitions(mininids PRIVATE "MININIDS_VERSION=\"${MININIDS_PACKAGE_LABEL}\"")
target_link_libraries(mininids PRIVATE mininids_core mininids_platform spdlog::spdlog)
if(APPLE AND MININIDS_BUILD_GUI)
  install(TARGETS mininids RUNTIME DESTINATION Mini-NIDS.app/Contents/MacOS)
else()
  set_target_properties(mininids PROPERTIES INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
  install(TARGETS mininids RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
