if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR WIN32)
  message(FATAL_ERROR "Fuzz targets require Linux/macOS Clang with libFuzzer")
endif()
target_compile_options(mininids_core PRIVATE -fsanitize=fuzzer-no-link,address,undefined)
foreach(kind IN ITEMS Pcap Dns Dispatcher Ipv4 Tcp)
  add_executable(fuzz_${kind} FuzzParsers.cpp)
  target_compile_definitions(fuzz_${kind} PRIVATE FUZZ_${kind}=1)
  target_compile_options(fuzz_${kind} PRIVATE -fsanitize=fuzzer,address,undefined -fno-omit-frame-pointer)
  target_link_options(fuzz_${kind} PRIVATE -fsanitize=fuzzer,address,undefined)
  target_link_libraries(fuzz_${kind} PRIVATE mininids_core)
endforeach()
